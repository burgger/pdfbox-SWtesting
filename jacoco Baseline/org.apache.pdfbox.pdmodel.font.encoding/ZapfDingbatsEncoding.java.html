     * Constructor.
     */

    {
        for (Object[] encodingEntry : ZAPFDINGBATS_ENCODING_TABLE)
        {
            add((Integer) encodingEntry[CHAR_CODE], encodingEntry[CHAR_NAME].toString());
        }
    }
    
    @Override
    public COSBase getCOSObject()
    {

    public String getEncodingName()
